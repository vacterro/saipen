var MSYS_DRIVE_PATTERN = /^\/([a-z])\//i, translatedPathCache = /* @__PURE__ */ new Map;
function translateGitBashPath(filePath, dependencies = {}) {
  if ((dependencies.platform ?? process.platform) !== "win32" || !isGitBashAbsolutePath(filePath))
    return filePath;
  if (MSYS_DRIVE_PATTERN.test(filePath))
    return `${filePath.charAt(1).toUpperCase()}:\\${filePath.slice(3).replace(/\//g, "\\")}`;
  let cached2 = translatedPathCache.get(filePath);
  if (cached2 !== void 0)
    return cached2;
  let env2 = dependencies.env ?? getSystemProcessEnv(), bash = (dependencies.findBash ?? findWindowsBash)(env2);
  if (!bash)
    return filePath;
  let cygpath = findCygpath(bash, dependencies.pathExists ?? pathExists);
  if (!cygpath)
    return filePath;
  let translated = runCygpath(dependencies.spawnCygpath ?? spawnCygpath, cygpath, filePath);
  if (translated === null)

  return translatedPathCache.set(filePath, translated), translated;
}
